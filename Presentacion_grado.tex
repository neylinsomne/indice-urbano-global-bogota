\section{Motivación}
% ================================================================

\begin{frame}{El problema: el precio NO mide calidad urbana}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
      \begin{itemize}
        \item En Bogotá, \hl{el precio inmobiliario refleja una mezcla} de
              ubicación, atributos físicos, inercia del mercado, percepción
              subjetiva, y \textit{shocks} especulativos.
        \item Los hogares y compradores carecen de una \hi{métrica objetiva}
              de la calidad del entorno urbano que no esté ya contaminada por
              el precio.
        \item Las herramientas comerciales asumen que ``más caro = mejor zona''
              --- un círculo lógico.
      \end{itemize}
    \column{0.4\textwidth}
      \includegraphics[width=\linewidth]{figs/intro_mapa.png}
  \end{columns}
\end{frame}
\note{\scriptsize
\textbf{Citas:} Khoshnoud et al. (2023); Shiller (2015).\par
\textbf{Puntos a mencionar:}
\begin{itemize}\itemsep0pt
  \item Arrancar la sustentación con la observación del problema: comprar/invertir en Bogotá se basa hoy en precios sin trazabilidad.
  \item ``Caro = bueno'' es un círculo lógico que el mercado refuerza con su propia inercia.
  \item Shiller (\textit{Irrational Exuberance}) discute justamente la falla de eficiencia cuando los fundamentales se desconectan del precio observado.
  \item Anticipar: vamos a construir una métrica que rompa el círculo.
\end{itemize}
}

\begin{frame}{Pregunta de investigación}
  \begin{block}{Hipótesis central}
    ¿Es posible construir una métrica \hi{ortogonal al precio} que sintetice
    la calidad urbana de Bogotá a partir de \textbf{datos oficiales abiertos},
    y que permita detectar \textit{mispricing} entre el valor de mercado y
    los fundamentales del entorno?
  \end{block}

  \vspace{0.8em}
  \begin{itemize}
    \item \textbf{Objetivo:} construir el Índice Urbano Global (IUG), una métrica
          \hi{[0, 5]} que combine 5 dimensiones del entorno urbano sin usar
          datos de mercado.
    \item \textbf{Aporte:} marco metodológico replicable + plataforma
          operacional + validación estadística completa.
  \end{itemize}
\end{frame}
\note{\scriptsize
\textbf{Citas:} Rosen (1974); Nardo et al. (2008).\par
\textbf{Puntos a mencionar:}
\begin{itemize}\itemsep0pt
  \item ``Ortogonalidad al precio'' es la clave: el indicador NO usa precio como insumo, así que tampoco lo predice trivialmente. Eso lo distingue del modelo hedónico clásico de Rosen.
  \item Tres entregables: metodología (defendible), plataforma (visible), validación (estadística).
  \item No es solo un score: es un objeto académico replicable porque sus insumos son datos abiertos oficiales.
\end{itemize}
}

\section{Marco teórico}
% ================================================================

\begin{frame}{Bases conceptuales}
  \begin{columns}[T,onlytextwidth]
    \column{0.5\textwidth}
      \textbf{Precios hedónicos.} Los atributos del bien explican parcialmente
      su precio; el resto se atribuye al entorno~\cite{rosen1974, Khoshnoud2023}.

      \vspace{0.5em}
      \textbf{Accesibilidad como masa potencial.} El acceso al transporte se
      modela como un sumatorio gravitacional~\cite{hansen1959, HaynesFotheringham1985}.

      \vspace{0.5em}
      \textbf{Indicadores compuestos.} OECD propone una metodología paso a
      paso (selección $\to$ normalización $\to$ ponderación $\to$
      validación)~\cite{nardo2008}.

    \column{0.5\textwidth}
      \textbf{Análisis espacial.} \textit{Everything is related to everything
      else, but near things are more related}~\cite{tobler1970}.
      Soporta el uso de buffers caminables (400/800~m) en
      el subíndice de dotación.

      \vspace{0.5em}
      \textbf{Multicolinealidad y commonality.} Los subíndices pueden compartir
      varianza; reportarla previene sobre-interpretación~\cite{nimon2013}.

      \vspace{0.5em}
      \textbf{Validación cruzada espacial.} Bloques espaciales evitan el
      ``optimism bias'' del CV aleatorio~\cite{roberts2017}.
  \end{columns}
\end{frame}
\note{\scriptsize
\textbf{Citas en este slide:} Rosen (1974); Khoshnoud et al. (2023);
Hansen (1959); Haynes \& Fotheringham (1985); Nardo et al. (2008);
Tobler (1970); Nimon \& Oswald (2013); Roberts et al. (2017).\par
\textbf{Discurso:}
\begin{itemize}\itemsep0pt
  \item Cada pilar teórico justifica UNA decisión del proyecto: hedonic justifica el modelo de precios, gravity justifica el cálculo de I\_ACC, OECD justifica la equiponderación, Tobler justifica los buffers caminables, commonality justifica reportar ortogonalidad, CV espacial justifica cómo validamos.
  \item Si el jurado pregunta ``¿por qué hicieron X?'', cada X aparece aquí con su soporte.
  \item Tobler 1970: la primera ley de geografía. Probablemente la cita más conocida en análisis espacial.
\end{itemize}
}

\begin{frame}{Brecha en la literatura}
  \begin{itemize}
    \item Modelos hedónicos clásicos: tratan al entorno como una variable
          de control, no como objeto de estudio.
    \item Indicadores compuestos urbanos: pocos se calculan a escala
          \textbf{de inmueble} (la mayoría a escala de barrio o localidad).
    \item Plataformas comerciales colombianas: precios sin trazabilidad
          metodológica.
    \item \hi{INMU} llena el espacio: indicador compuesto, a nivel de
          inmueble, ortogonal al precio, calculado con datos oficiales,
          servido vía API y validado estadísticamente.
  \end{itemize}
\end{frame}
\note{\scriptsize
\textbf{Puntos a mencionar:}
\begin{itemize}\itemsep0pt
  \item Los hedónicos miden \emph{coeficientes}, no construyen un indicador.
  \item Indicadores urbanos suelen ser a escala de barrio (CCB) o ciudad (ONU-Habitat). A escala de INMUEBLE es lo nuevo.
  \item Plataformas comerciales (Metrocuadrado, FincaRaíz) NO publican metodología — sus precios son ``de mercado'', no fundamentales.
  \item INMU = único proyecto que combina las cuatro cosas: a nivel de inmueble, abierto, validado, operacional.
\end{itemize}
}

\section{Metodología · Las 5 dimensiones del IUG}
% ================================================================

\begin{frame}{Estructura del IUG}
  \[
    \mathrm{IUG}_i \;=\; w_{\text{acc}} \, I_{\text{ACC},i}
                  \,+\, w_{\text{seg}} \, I_{\text{SEG},i}
                  \,+\, w_{\text{hed}} \, I_{\text{HED},i}
                  \,+\, w_{\text{dot}} \, I_{\text{DOT},i}
                  \,+\, w_{\text{pnu}} \, I_{\text{PNU},i}
  \]
  \begin{center}
    \footnotesize Cada subíndice $I_{\bullet,i} \in [0, 5]$ tras
    \textit{rank min-max} y pesos equiponderados por defecto (0.20 c/u).
  \end{center}
  \vspace{0.4em}
  \begin{block}{Premisa}
    Los datos de entrada son \emph{independientes} de la oferta inmobiliaria:
    DANE, IDECA, SDP, EPV 2024, POT 555, IDRD. Esto garantiza la
    \hi{ortogonalidad al precio} que validamos estadísticamente.
  \end{block}
  \framenote{Equiponderación 0.20 c/u sigue la recomendación OECD para
  indicadores compuestos cuando no hay información a priori sobre la
  importancia relativa de los componentes~\cite{nardo2008}. Discusión sobre
  compensatoriedad en la agregación lineal~\cite{munda2008}.}
\end{frame}
\note{\scriptsize
\textbf{Citas:} Nardo et al. (2008) OECD; Munda (2008).\par
\textbf{Discurso:}
\begin{itemize}\itemsep0pt
  \item Combinación lineal compensatoria: una zona puede compensar baja accesibilidad con buena dotación. Es lo que Munda llama \emph{compensatoriedad parcial}.
  \item Si jurado pregunta ``¿por qué no GeoMean?'': discutirlo. GeoMean penaliza más los ceros, pero introduce supuesto fuerte de no-compensatoriedad que aquí no aplica.
  \item Datos abiertos = replicabilidad. Cualquier persona puede recalcular el IUG en su PC con los scripts del repo.
\end{itemize}
}

\begin{frame}{¿Por qué estos pesos? AHP + Equiponderación}
  Decisión metodológica en \hi{dos niveles}:

  \vspace{0.6em}
  \textbf{1. Entre los 5 subíndices (nivel macro):}\quad
  Pesos \hi{equiponderados $w_k = 0.20$}.

  \begin{itemize}
    \item Sin evidencia empírica que justifique a priori que una dimensión
          urbana ``pesa'' más que otra para todos los hogares.
    \item Recomendación explícita del \emph{Handbook} de indicadores
          compuestos de la OECD~\cite{nardo2008}.
    \item Evita supuestos arbitrarios y permite que el lector aplique
          sus propios pesos (el sistema acepta vectores custom).
  \end{itemize}

  \vspace{0.4em}
  \textbf{2. Dentro de un subíndice (nivel micro):}\quad
  Pesos derivados por \hi{AHP} cuando hay juicio experto sólido.

  \begin{itemize}
    \item Caso aplicado: I\_SEG combina 7 tipos de delito ponderados
          por \emph{severidad} (homicidio 0.566, delito sexual 0.267,
          hurto 0.120, etc.).
    \item AHP ofrece matriz de comparaciones pareadas + chequeo de
          consistencia ($CR < 0.10$).
  \end{itemize}
  \framenote{AHP original de Saaty~\cite{saaty1980}; revisión moderna y
  cuándo aplicarlo en sistemas compuestos~\cite{ishizaka2011}.
  Equiponderación entre dimensiones~\cite{nardo2008}; discusión
  compensatoriedad~\cite{munda2008}.}
\end{frame}
\note{\scriptsize
\textbf{Citas:} Saaty (1980); Ishizaka \& Labib (2011); Nardo et al. (2008).\par
\textbf{Si te preguntan ``¿por qué 0.20 c/u, no es arbitrario?'':}
\begin{itemize}\itemsep0pt
  \item No es arbitrario: es la regla de \emph{máxima entropía} entre los criterios cuando no hay información a priori. La justifica OECD explícitamente.
  \item Si pones uno mayor que otro estás INTRODUCIENDO una preferencia que NO existe en los datos.
  \item El sistema acepta vectores custom: un evaluador catastral puede aplicar sus propios pesos y recalcular en segundos.
\end{itemize}
\textbf{Si te preguntan ``¿por qué AHP solo dentro de I\_SEG?'':}
\begin{itemize}\itemsep0pt
  \item Porque ahí SÍ hay juicio normativo claro: la sociedad colombiana considera el homicidio más grave que el hurto a residencia. Eso se puede capturar con AHP.
  \item Entre las 5 dimensiones (accesibilidad vs. seguridad vs. ...) NO hay un juicio normativo universal — depende del hogar.
  \item Saaty: AHP requiere comparaciones pareadas con consistencia CR < 0.10. La validamos.
\end{itemize}
}

\begin{frame}{I\_ACC · Accesibilidad}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
      Modelo gravitacional sobre TransMilenio + SITP:
      \[
        I_{\text{ACC},i} = \sum_{j \in J} \frac{O_j}{d_{ij}^{\beta}}
      \]
      \begin{itemize}
        \item $O_j$: capacidad/jerarquía de la estación.
        \item $d_{ij}$: distancia geodésica desde el inmueble.
        \item $\beta$: parámetro de decaimiento espacial \\
              ($\beta \approx 1.5$ tras calibración).
      \end{itemize}
      \vspace{0.4em}
      Fuentes: TMSA (149 estaciones), SITP-OSM (7\,693 paraderos).
    \column{0.4\textwidth}
      \includegraphics[width=\linewidth]{figs/hist_iacc.pdf}
  \end{columns}
  \framenote{Modelo gravitacional clásico de masa potencial de
  Hansen~\cite{hansen1959}, generalizado al marco de interacción
  espacial por Haynes \& Fotheringham~\cite{HaynesFotheringham1985}.
  El decaimiento por distancia es consistente con la primera ley de
  geografía de Tobler~\cite{tobler1970}.}
\end{frame}
\note{\scriptsize
\textbf{Citas:} Hansen (1959); Haynes \& Fotheringham (1985); Tobler (1970).\par
\textbf{Discurso:}
\begin{itemize}\itemsep0pt
  \item Hansen 1959 es el paper fundador de accesibilidad. Antes era ``distancia mínima'' — Hansen propuso sumar SOBRE todas las oportunidades pesadas por distancia. Esa idea de ``masa potencial'' es lo que usamos.
  \item $\beta = 1.5$ es calibrado: se hizo grid search sobre el ajuste R² de log(precio) y ese valor maximiza la relación monótona con el precio observado.
  \item TM: 149 estaciones. SITP: 7.693 paraderos. Esto da cobertura espacial real.
\end{itemize}
}

\begin{frame}{I\_SEG · Seguridad tridimensional}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
      Tres capas combinadas:
      \[
        I_{\text{SEG},i}' = 0.40\,S_{\text{micro}} + 0.30\,D_{\text{obj}}
                          + 0.30\,D_{\text{subj}}
      \]
      \begin{itemize}
        \item $S_{\text{micro}}$: proximidad a CAI (599 puntos).
        \item $D_{\text{obj}}$: criminalidad por km$^2$ (ICSU/SIEDCO).
        \item $D_{\text{subj}}$: percepción ciudadana \\
              EPV 2024 ($n = 19\,354$)~\cite{ccb2024}.
      \end{itemize}
      \vspace{0.4em}
      \textit{Paradoja de la inseguridad}: el 44\% de localidades
      tienen criminalidad alta y percepción media --- la dimensión
      objetiva sola no basta.
    \column{0.4\textwidth}
      \includegraphics[width=\linewidth]{figs/scatter_seg.pdf}
  \end{columns}
  \framenote{Pesos por severidad de delito mediante \emph{Analytic
  Hierarchy Process}~\cite{saaty1980, ishizaka2011}: homicidio 56.6\%,
  delitos sexuales 26.7\%, hurto a personas 12\%. Percepción ciudadana de
  la EPV 2024 ($n = 19\,354$)~\cite{ccb2024}. Marco de seguridad urbana
  multidimensional~\cite{cardenas2017}.}
\end{frame}
\note{\scriptsize
\textbf{Citas:} Saaty (1980); Ishizaka \& Labib (2011); CCB (2024); Cárdenas (2017).\par
\textbf{Discurso:}
\begin{itemize}\itemsep0pt
  \item Las tres capas miden cosas \emph{distintas} pero correlacionadas:
  \begin{itemize}
    \item Micro (proximidad CAI) = sentirse seguro porque hay policía cerca.
    \item Objetiva (ICSU) = lo que dicen los reportes de SIEDCO.
    \item Subjetiva (EPV) = cómo se sienten realmente los 19.354 encuestados.
  \end{itemize}
  \item \emph{Paradoja de la inseguridad}: hay localidades con baja criminalidad objetiva pero alta percepción de inseguridad (y viceversa). La métrica pura de denuncias falla — por eso necesitamos las tres capas.
  \item AHP da los pesos DENTRO de la dimensión objetiva: homicidio pesa más que hurto porque la sociedad lo considera más grave.
\end{itemize}
}

\begin{frame}{I\_HED · Hedónico (PCA validado)}
  \begin{columns}[T,onlytextwidth]
    \column{0.6\textwidth}
      Score derivado de \hi{Análisis de Componentes Principales}
      sobre área construida, baños, garajes, ascensor, antigüedad.

      \vspace{0.4em}
      Validación previa al PCA~\cite{hair2010}:
      \begin{itemize}
        \item KMO $\geq 0.50$~\cite{kaiser1974}.
        \item Bartlett $p < 0.05$~\cite{bartlett1954}.
        \item Bootstrap de 200 réplicas para estabilidad de cargas.
      \end{itemize}
    \column{0.35\textwidth}
      \includegraphics[width=\linewidth]{figs/scree_ihed.pdf}
  \end{columns}
  \framenote{Marco hedónico clásico de Rosen para atribuir parcialmente
  el precio a atributos del bien~\cite{rosen1974}; evolución contemporánea
  de las técnicas hedónicas en~\cite{Khoshnoud2023}. Validación PCA
  estándar Hair et~al.~\cite{hair2010} con índices de Kaiser
  (KMO)~\cite{kaiser1974} y test de esfericidad de
  Bartlett~\cite{bartlett1954}.}
\end{frame}
\note{\scriptsize
\textbf{Citas:} Rosen (1974); Khoshnoud et al. (2023); Hair et al. (2010); Kaiser (1974); Bartlett (1954).\par
\textbf{Discurso:}
\begin{itemize}\itemsep0pt
  \item I\_HED captura el ``confort privado'': lo que el inmueble ofrece a quien vive ADENTRO. Es la única dimensión que NO depende del entorno.
  \item PCA reduce 5--6 variables correlacionadas a un score único. Usamos PC1 que típicamente explica $>$60\% de varianza.
  \item Validación previa al PCA es CRÍTICA: KMO $\geq$ 0.50 (mínimo aceptable) y Bartlett $p < 0.05$ (las variables comparten varianza). Si KMO falla, el PCA es inválido.
  \item Bootstrap de 200 réplicas: confirma que las cargas factoriales son estables, no fruto de azar muestral.
\end{itemize}
}

\begin{frame}{I\_DOT · Dotación de equipamientos}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
      Proximidad acumulada con radios de caminabilidad:
      \[
        I_{\text{DOT},i} = \sum_{c} w_c \!\left(
          |\text{POI}_c|_{400m} + 0.5\,|\text{POI}_c|_{800m}
        \right)
      \]
      Categorías (\emph{post-corpus v2.0}):
      \begin{itemize}
        \item Salud: 8\,369 (IPS + farmacias)
        \item Educación: 2\,820 (colegios + univ.)
        \item Cultura: 157 (bibliotecas + teatros)
        \item Comercio: 70 (CC)
        \item Recreación: 194 (parques IDRD + escenarios)
      \end{itemize}
    \column{0.4\textwidth}
      \includegraphics[width=\linewidth]{figs/poi_categorias.pdf}
  \end{columns}
  \framenote{Buffers caminables de 400/800~m son estándar internacional
  de \textit{walkability} en planeación urbana; alineados con la primera
  ley de Tobler~\cite{tobler1970}. Pesos por categoría según peso urbano
  típico en el \emph{Handbook} de indicadores compuestos
  OECD~\cite{nardo2008}.}
\end{frame}
\note{\scriptsize
\textbf{Citas:} Tobler (1970); Nardo et al. (2008).\par
\textbf{Discurso:}
\begin{itemize}\itemsep0pt
  \item 400~m = 5 min caminando a 1.3 m/s (velocidad peatonal promedio). 800~m = 10 min. Son los radios estándar de \emph{walkability} de la OECD y la New Urban Agenda.
  \item Los POIs dentro de 400~m valen el doble que los entre 400--800~m. Refleja que el acceso inmediato es cualitativamente distinto.
  \item Crítico: aquí cargamos POST-entrega los datasets de colegios SED, parques IDRD, etc. Pasamos de 4 categorías a 10 (versión v2.0). Ver slide de volumetría.
\end{itemize}
}

\begin{frame}{I\_PNU · Potencial normativo (POT 555)}
  \[
    I_{\text{PNU},i} = 0.4\,\text{score}_\text{trat} + 0.4\,\text{score}_\text{edif}
                     + 0.2\,\text{score}_\text{uso}
  \]
  \begin{itemize}
    \item Tratamiento urbanístico (5\,703 polígonos): consolidación,

\begin{frame}{Estudio de normalidad (subíndices, n = 23\,761)}
  \footnotesize
  \begin{center}
  \begin{tabular}{lrrrl}
    \toprule
    \textbf{Subíndice} & \textbf{Skewness} & \textbf{Kurtosis exc.} &
    \textbf{Shapiro--Wilk W} & \textbf{Veredicto} \\
    \midrule
    I\_URB             &  $-0.42$ &  $-0.71$ & 0.945 & no normal (asimétrico) \\
    I\_ACC             &  $+0.13$ &  $-1.65$ & 0.830 & \hi{bimodal} \\
    I\_SEG             &  $-0.26$ &  $-0.57$ & 0.915 & no normal \\
    I\_HED             &  $-0.62$ &  $-0.48$ & 0.890 & asimétrico izquierdo \\
    I\_DOT             &  $-0.26$ &  $-1.66$ & 0.797 & \hi{bimodal} \\
    I\_PNU             &  $+0.31$ &  $-0.29$ & 0.891 & casi simétrico \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.4em}
  \footnotesize
  \textbf{Lectura:} ningún subíndice pasa las pruebas formales de
  normalidad (Shapiro--Wilk, Jarque--Bera, Anderson--Darling, K--S
  rechazan $H_0$ con $p < 10^{-40}$). \hi{Esto es esperable y
  académicamente correcto}: los subíndices están normalizados a [0, 5]
  por \textit{rank min-max}, lo que produce distribuciones con soporte
  acotado, frecuentemente bimodales. La inferencia se hace por
  \emph{intervalos bootstrap} y t-Student conforme al protocolo
  IAAO~\cite{iaao2013, student1908}.
\end{frame}
\note{\scriptsize
\textbf{Citas:} IAAO (2013); Student/Gosset (1908).\par
\textbf{Si te preguntan ``¿por qué no son normales?'':}
\begin{itemize}\itemsep0pt
  \item Porque están normalizados a [0, 5] por \emph{rank min-max}, lo que produce soporte ACOTADO. Una distribución normal tiene soporte $(-\infty, \infty)$.
  \item Los subíndices I\_ACC e I\_DOT son \emph{bimodales} (kurtosis exceso $\approx -1.65$): refleja que muchos inmuebles están o muy cerca o muy lejos de TM/equipamientos. No hay punto medio en Bogotá.
  \item Eso NO es un problema: rechazar normalidad no significa rechazar el indicador. Solo significa que NO podemos usar inferencia paramétrica clásica $\Rightarrow$ usamos bootstrap.
  \item IAAO 2013 es el estándar de tasación masiva internacional. Recomienda explícitamente bootstrap cuando hay no-normalidad.
\end{itemize}
}

\begin{frame}{Normalidad de residuos hedónicos (n = 11\,539)}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
      \footnotesize
      Modelo: $\log(P_i) = \beta_0 + \beta_1 \text{IUG} + \beta_2 \log A_i
                          + \beta_3 \text{hab}_i + \beta_4 \text{baños}_i
                          + \beta_5 \text{estrato}_i + \varepsilon_i$

      \vspace{0.5em}
      \begin{tabular}{lr}
        \toprule
        \textbf{Estadístico}      & \textbf{Valor}    \\
        \midrule
        Skewness                  & $-3.68$           \\
        Kurtosis exceso           & $+38.96$          \\
        Jarque--Bera $\chi^2$     & 755\,643          \\
        Anderson--Darling $A^2$   & 273.3 (crit 0.75) \\
        Veredicto                 & \hi{colas pesadas} \\
        \bottomrule
      \end{tabular}
    \column{0.4\textwidth}
      \includegraphics[width=\linewidth]{figs/qq_residuos.pdf}
  \end{columns}
  \vspace{0.6em}
  \footnotesize
  Los residuos exhiben colas pesadas (kurtosis exceso $\approx 39$):
  evidencia de \emph{outliers} legítimos en el mercado bogotano (lujo,
  oportunidades). \hi{La inferencia se hace por bootstrap y errores
  estándar robustos} (HC3), no por aproximación normal asintótica.
\end{frame}
\note{\scriptsize
\textbf{Discurso:}
\begin{itemize}\itemsep0pt
  \item Skewness $-3.68$ + kurtosis exceso $+39$: residuos con colas \emph{muy} pesadas. Es lo esperable en mercado inmobiliario: hay outliers reales (penthouses de lujo, oportunidades de inversión).
  \item NO es un error del modelo — son inmuebles reales que el modelo no puede capturar con un solo coeficiente lineal.
  \item HC3 (Heteroskedasticity-Consistent type 3): corrección estándar para errores robustos. Recomendada cuando hay heterocedasticidad o no-normalidad.
  \item Si jurado insiste: GWR (geographically weighted regression) podría capturar mejor los outliers locales. Está en ``trabajo futuro''.
\end{itemize}
}

\begin{frame}{Ortogonalidad entre subíndices}
  \begin{center}
    \includegraphics[width=0.95\linewidth]{figs/heatmap_corr.pdf}
  \end{center}
  \vspace{-0.6em}
  \footnotesize
  \hi{Lectura:} ningún par de subíndices supera $|r| > 0.5$ ni $|\rho| > 0.5$,
  confirmando que cada dimensión aporta información independiente.
  Esto valida la \textbf{premisa de ortogonalidad} del IUG y justifica
  reportar los 5 subíndices por separado --- el agregado no captura el
  detalle accionable.
\end{frame}
\note{\scriptsize
\textbf{Citas:} Nimon \& Oswald (2013); Anselin (1995).\par
\textbf{Discurso:}
\begin{itemize}\itemsep0pt
  \item Esta es la prueba VISUAL de que las 5 dimensiones NO redundan: si la correlación entre I\_ACC e I\_DOT fuera $r=0.9$, sería absurdo tenerlas separadas.
  \item Que ningún par supere $|r|>0.5$ valida el diseño: cada dimensión es genuinamente independiente.
  \item Spearman (no-paramétrico) confirma lo mismo que Pearson — robustez a no-normalidad.
  \item Nimon \& Oswald 2013 desarrollaron precisamente la idea de \emph{commonality analysis} para descomponer varianza compartida en indicadores compuestos.
\end{itemize}
}

\end{itemize}

\begin{frame}{Aportes}
  \begin{itemize}
    \item \hi{Metodológico:} indicador compuesto a nivel de inmueble,
          construido por OECD-style con validación estadística completa
          (commonality, bootstrap, CV espacial)~\cite{nardo2008, roberts2017}.
    \item \hi{Empírico:} 23\,761 inmuebles bogotanos georreferenciados,
          11\,610 POIs cargados, 5\,703 polígonos POT. Trazabilidad
          v1.0 $\to$ v2.0 documentada.
    \item \hi{Tecnológico:} plataforma operacional en producción,
          API documentada, hábeas data conforme a Ley 1581/2012,
          repositorio público.
    \item \hi{Social:} insumo objetivo para hogares y reguladores ---
          el IUG es libre de la inercia del mercado.
  \end{itemize}
\end{frame}
\note{\scriptsize
\textbf{Citas:} Nardo et al. (2008); Roberts et al. (2017).\par
\textbf{Resumen de los cuatro aportes (uno por verticales):}
\begin{itemize}\itemsep0pt
  \item METODOLÓGICO: el ``cómo'' replicable.
  \item EMPÍRICO: las 23.761 observaciones que sostienen el análisis.
  \item TECNOLÓGICO: la plataforma que demuestra que es operacional, no solo teórico.
  \item SOCIAL: el ``para qué'' --- democratizar evaluación urbana objetiva.
\end{itemize}
}

\begin{frame}{Limitaciones y trabajo futuro}
  \textbf{Limitaciones reconocidas:}
  \begin{itemize}
    \item Cobertura del corpus restringida a Bogotá (multi-ciudad pendiente).
    \item Plazas de mercado (42 polígonos) con CRS no estándar --- pendiente
          de cargar.
    \item Residuos hedónicos con colas pesadas: requiere errores robustos
          o bootstrap (lo hicimos).
  \end{itemize}
  \vspace{0.6em}
  \textbf{Líneas futuras:}
  \begin{itemize}
    \item Modelo hedónico con regresión cuantílica~\cite{koenker2005}
          para evaluar el efecto del IUG en distintos segmentos del precio.
    \item GWR (\textit{geographically weighted regression}) con
          diagnóstico de colinealidad~\cite{wheeler2007}.
    \item Series temporales del IUG (pipeline mensual ya implementado,
          falta análisis longitudinal).
  \end{itemize}
\end{frame}
\note{\scriptsize
\textbf{Citas:} Koenker (2005); Wheeler (2007).\par
\textbf{Sé honesto con las limitaciones — el jurado lo valora:}
\begin{itemize}\itemsep0pt
  \item Multi-ciudad: tenemos 21.671 inmuebles de Medellín/Cali/Barranquilla en la BD, pero las capas POT/TM son solo de Bogotá. Los marcamos con \texttt{iurb=NULL} para no contaminar promedios. Reto: replicar el pipeline para otras ciudades requiere cargar sus capas oficiales.
  \item Plazas de mercado: 42 polígonos del IPES con CRS no estándar. Impacto marginal en I\_DOT.
  \item Colas pesadas en residuos: $\Rightarrow$ trabajos futuros con regresión cuantílica (Koenker 2005) o GWR (Wheeler 2007).
\end{itemize}
\textbf{Si te preguntan ``¿qué harías diferente si empezaras hoy?'':}
\begin{itemize}\itemsep0pt
  \item Cargar TODOS los datasets antes de empezar el análisis (lección aprendida del v1.0 $\to$ v2.0).
  \item Diseñar el sistema multi-ciudad desde el inicio.
\end{itemize}
